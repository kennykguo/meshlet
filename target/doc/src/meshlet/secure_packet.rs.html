1//! The encrypted data packet used after a successful peer handshake.
2//!
3//! A small visible header carries the direction and packet number. The payload
4//! is encrypted, and ChaCha20-Poly1305 also authenticates the visible header.
5
6use chacha20poly1305::{
7    ChaCha20Poly1305, Key, KeyInit, Nonce,
8    aead::{Aead, Payload},
9};
10
11const MAGIC: [u8; 4] = *b"MSH3";
12const HEADER_BYTES: usize = 13;
13const AUTHENTICATION_TAG_BYTES: usize = 16;
14
15#[derive(Clone, Copy, Eq, PartialEq)]
16pub(crate) enum Direction {
17    ClientToServer,
18    ServerToClient,
19}
20
21impl Direction {
22    /// Converts a packet direction into its one-byte wire representation.
23    /// Called while building packet headers and nonces.
24    fn wire_value(self) -> u8 {
25        match self {
26            Self::ClientToServer => 0,
27            Self::ServerToClient => 1,
28        }
29    }
30
31    /// Parses a one-byte wire direction or rejects unknown values.
32    /// Called by `PacketReceiver::open` before decrypting a packet.
33    fn from_wire(value: u8) -> Result<Self, String> {
34        match value {
35            0 => Ok(Self::ClientToServer),
36            1 => Ok(Self::ServerToClient),
37            _ => Err(format!("unknown encrypted-packet direction {value}")),
38        }
39    }
40
41    /// Returns a human-readable direction for validation errors.
42    /// Called by `PacketReceiver::open` when the received direction is wrong.
43    fn label(self) -> &'static str {
44        match self {
45            Self::ClientToServer => "client-to-server",
46            Self::ServerToClient => "server-to-client",
47        }
48    }
49}

51pub(crate) struct PacketSender {
52    cipher: ChaCha20Poly1305,
53    direction: Direction,
54    next_packet_number: u64,
55}
56
57impl PacketSender {
58    /// Creates an encryptor for one key and one traffic direction at packet zero.
59    /// Called by secure echo clients and servers after session-key derivation.
60    pub(crate) fn new(key: [u8; 32], direction: Direction) -> Self {
61        let key = Key::from(key);
62        Self {
63            cipher: ChaCha20Poly1305::new(&key),
64            direction,
65            next_packet_number: 0,
66        }
67    }
68
69    /// Encrypts and authenticates one plaintext under the next packet number.
70    /// Called by secure echo clients for requests and servers for responses.
71    pub(crate) fn seal(&mut self, plaintext: &[u8]) -> Result<Vec<u8>, String> {
72        let packet_number = self.next_packet_number;
73        let next_packet_number = packet_number
74            .checked_add(1)
75            .ok_or_else(|| "outbound packet number exhausted".to_string())?;
76        let header = packet_header(self.direction, packet_number);
77        let nonce = packet_nonce(self.direction, packet_number);
78        let ciphertext = self
79            .cipher
80            .encrypt(
81                &nonce,
82                Payload {
83                    msg: plaintext,
84                    // AAD stays visible but is covered by the authentication tag.
85                    aad: &header,
86                },
87            )
88            .map_err(|_| "failed to encrypt packet".to_string())?;
89
90        self.next_packet_number = next_packet_number;
91
92        let mut packet = Vec::with_capacity(header.len() + ciphertext.len());
93        packet.extend_from_slice(&header);
94        packet.extend_from_slice(&ciphertext);
95        Ok(packet)
96    }
97}

99pub(crate) struct PacketReceiver {
100    cipher: ChaCha20Poly1305,
101    direction: Direction,
102    next_packet_number: u64,
103}
104
105impl PacketReceiver {
106    /// Creates a decryptor expecting one key, direction, and packet number zero.
107    /// Called by secure echo clients and servers after session-key derivation.
108    pub(crate) fn new(key: [u8; 32], direction: Direction) -> Self {
109        let key = Key::from(key);
110        Self {
111            cipher: ChaCha20Poly1305::new(&key),
112            direction,
113            next_packet_number: 0,
114        }
115    }
116
117    /// Validates ordering and authentication before returning decrypted plaintext.
118    /// Called by secure echo clients for responses and servers for requests.
119    pub(crate) fn open(&mut self, packet: &[u8]) -> Result<Vec<u8>, String> {
120        if packet.len() < HEADER_BYTES + AUTHENTICATION_TAG_BYTES {
121            return Err("encrypted packet is too short".into());
122        }
123        if packet[..MAGIC.len()] != MAGIC {
124            return Err("encrypted packet has the wrong protocol marker".into());
125        }
126
127        let direction = Direction::from_wire(packet[4])?;
128        if direction != self.direction {
129            return Err(format!(
130                "expected a {} packet, received {}",
131                self.direction.label(),
132                direction.label()
133            ));
134        }
135
136        let packet_number = u64::from_be_bytes(
137            packet[5..HEADER_BYTES]
138                .try_into()
139                .expect("packet-number header has a fixed size"),
140        );
141        if packet_number != self.next_packet_number {
142            return Err(format!(
143                "expected packet number {}, received {packet_number}",
144                self.next_packet_number
145            ));
146        }
147        let next_packet_number = packet_number
148            .checked_add(1)
149            .ok_or_else(|| "inbound packet number exhausted".to_string())?;
150
151        let nonce = packet_nonce(direction, packet_number);
152        let plaintext = self
153            .cipher
154            .decrypt(
155                &nonce,
156                Payload {
157                    msg: &packet[HEADER_BYTES..],
158                    aad: &packet[..HEADER_BYTES],
159                },
160            )
161            .map_err(|_| "encrypted packet authentication failed".to_string())?;
162
163        self.next_packet_number = next_packet_number;
164        Ok(plaintext)
165    }
166}

168/// Encodes the protocol marker, direction, and packet number into a visible header.
169/// Called by `PacketSender::seal` before authenticated encryption.
170fn packet_header(direction: Direction, packet_number: u64) -> [u8; HEADER_BYTES] {
171    let mut header = [0_u8; HEADER_BYTES];
172    header[..MAGIC.len()].copy_from_slice(&MAGIC);
173    header[4] = direction.wire_value();
174    header[5..].copy_from_slice(&packet_number.to_be_bytes());
175    header
176}
177
178/// Derives a unique 96-bit ChaCha20-Poly1305 nonce from direction and packet number.
179/// Called by packet sealing and opening so both peers reconstruct the same nonce.
180fn packet_nonce(direction: Direction, packet_number: u64) -> Nonce {
181    let mut nonce = [0_u8; 12];
182    nonce[0] = direction.wire_value();
183    nonce[4..].copy_from_slice(&packet_number.to_be_bytes());
184    Nonce::from(nonce)
185}

187#[cfg(test)]
188mod tests {
189    use super::*;
190
191    /// Verifies successful decryption and rejection of a repeated packet number.
192    /// Called automatically by Rust's test harness during `cargo test`.
193    #[test]
194    fn encrypted_packet_round_trips_and_replay_is_rejected() {
195        let key = [7_u8; 32];
196        let mut sender = PacketSender::new(key, Direction::ClientToServer);
197        let mut receiver = PacketReceiver::new(key, Direction::ClientToServer);
198        let packet = sender.seal(b"secret payload").unwrap();
199
200        assert_eq!(receiver.open(&packet).unwrap(), b"secret payload");
201        assert!(receiver.open(&packet).is_err());
202    }
203
204    /// Verifies failed authentication does not consume the expected packet number.
205    /// Called automatically by Rust's test harness during `cargo test`.
206    #[test]
207    fn modified_ciphertext_is_rejected_without_advancing_receiver() {
208        let key = [9_u8; 32];
209        let mut sender = PacketSender::new(key, Direction::ServerToClient);
210        let mut receiver = PacketReceiver::new(key, Direction::ServerToClient);
211        let packet = sender.seal(b"secret payload").unwrap();
212        let mut modified = packet.clone();
213        let last = modified.last_mut().unwrap();
214        *last ^= 1;
215
216        assert!(receiver.open(&modified).is_err());
217        assert_eq!(receiver.open(&packet).unwrap(), b"secret payload");
218    }
